// SETU — "admin-create-user" edge function // Lets an administrator create an account directly, instead of the person self-registering. // Two modes: // password → creates the account with a temporary password the admin passes on; the // person is forced to choose their own at first sign-in // invite → emails the person a link to set their own password (needs working email; // see the SMTP note in the README) // The service-role key never leaves this function. import { createClient } from "npm:@supabase/supabase-js@2"; const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const reply = (status: number, body: unknown) => new Response(JSON.stringify(body), { status, headers: { ...CORS, "content-type": "application/json" } }); Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS }); if (req.method !== "POST") return reply(405, { error: "POST only" }); const url = Deno.env.get("SUPABASE_URL")!, service = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const admin = createClient(url, service, { auth: { persistSession: false } }); // who is asking? (the app sends the signed-in person's token) const jwt = (req.headers.get("authorization") || "").replace(/^Bearer\s+/i, ""); const { data: who } = await admin.auth.getUser(jwt); if (!who?.user) return reply(401, { error: "Sign in first" }); const { data: caller } = await admin.from("profiles") .select("is_admin,is_owner,active,name,can_create_users,can_manage_users").eq("id", who.user.id).single(); if (!caller?.active) return reply(403, { error: "Sign in first" }); const fullAdmin = !!caller.is_admin; // full admins (the owner is one) const mayCreate = fullAdmin || !!caller.can_create_users; // "create users" power const mayManage = fullAdmin || !!caller.can_manage_users; // "manage users" power (password reset) let b: any; try { b = await req.json(); } catch { return reply(400, { error: "Bad request" }); } // ---- reset an existing person's password (admin does it for someone locked out) ---- if (b.action === "reset_password") { if (!mayManage) return reply(403, { error: "You do not have access to manage users" }); const target = String(b.user_id || ""); if (!target) return reply(400, { error: "No account given" }); const { data: tgt } = await admin.from("profiles").select("name,is_owner,is_admin").eq("id", target).single(); if (!tgt) return reply(400, { error: "Account not found" }); if (tgt.is_owner && target !== who.user.id) return reply(403, { error: "Only the system owner can reset their own password" }); if (!fullAdmin && tgt.is_admin) return reply(403, { error: "Only a full administrator can reset an administrator's password" }); const typed = String(b.password || "").trim(); let pw = typed; if (typed) { if (typed.length < 8) return reply(400, { error: "A temporary password needs at least 8 characters" }); } else { const abc = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789"; const bytes = crypto.getRandomValues(new Uint8Array(10)); pw = Array.from(bytes, (x) => abc[x % abc.length]).join(""); } const { error: pErr } = await admin.auth.admin.updateUserById(target, { password: pw }); if (pErr) return reply(400, { error: pErr.message }); await admin.from("profiles").update({ must_change_password: true }).eq("id", target); await admin.from("audit").insert({ request_id: null, ref: null, actor_id: who.user.id, actor_name: caller.name, action: "Reset a password", detail: `${tgt.name}` }); return reply(200, { reset: true, name: tgt.name, tempPassword: pw }); } if (!mayCreate) return reply(403, { error: "You do not have access to create users" }); const name = String(b.name || "").trim(), email = String(b.email || "").trim().toLowerCase(); const role = String(b.role || "").trim(), dept = String(b.dept || "").trim(); const mode = b.mode === "invite" ? "invite" : "password"; if (name.length < 3) return reply(400, { error: "Enter the person's full name" }); if (!/^\S+@\S+\.\S+$/.test(email)) return reply(400, { error: "Enter a valid work email" }); if (role.length < 2) return reply(400, { error: "Enter a designation" }); const makeAdmin = !!b.is_admin; if (makeAdmin && !caller.is_owner) return reply(403, { error: "Only the system owner can create administrators" }); // Only a full admin may mint a sub-admin; a sub-admin creating users makes ordinary users only. const subCreate = fullAdmin && !!b.can_create_users && !makeAdmin; const subManage = fullAdmin && !!b.can_manage_users && !makeAdmin; // admins are a system role, not a step in the flow: a department is optional for them if (dept) { const { data: deptRow } = await admin.from("departments").select("name").eq("name", dept).maybeSingle(); if (!deptRow) return reply(400, { error: "Choose a department from the list" }); } else if (!makeAdmin) return reply(400, { error: "Choose a department from the list" }); const meta = { name, role, dept, manager_id: b.manager_id || "" }; let userId: string, tempPassword: string | null = null; if (mode === "invite") { const { data, error } = await admin.auth.admin.inviteUserByEmail(email, { data: meta, redirectTo: Deno.env.get("APP_URL") || undefined, }); if (error) return reply(400, { error: /already/i.test(error.message) ? "That email is already registered" : error.message }); userId = data.user.id; } else { const chosen = String(b.password || "").trim(); if (chosen && chosen.length < 8) return reply(400, { error: "A temporary password needs at least 8 characters" }); if (chosen) tempPassword = chosen; else { // 10 characters, no look-alike letters, easy to read out over the phone const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789"; const bytes = crypto.getRandomValues(new Uint8Array(10)); tempPassword = Array.from(bytes, (x) => alphabet[x % alphabet.length]).join(""); } const { data, error } = await admin.auth.admin.createUser({ email, password: tempPassword, email_confirm: true, user_metadata: meta, }); if (error) return reply(400, { error: /already/i.test(error.message) ? "That email is already registered" : error.message }); userId = data.user.id; } // the profile row was created by the database trigger; apply the access flags on top const { error: upErr } = await admin.from("profiles").update({ is_manager: !!b.is_manager, see_all: (fullAdmin && !!b.see_all) || makeAdmin, is_admin: makeAdmin, can_create_users: subCreate, can_manage_users: subManage, manager_id: b.manager_id || null, manager_confirmed: !!b.manager_id, must_change_password: true, pages: Array.isArray(b.pages) ? b.pages : null, }).eq("id", userId); if (upErr) return reply(500, { error: "Account created but access could not be set: " + upErr.message }); const roleLabel = makeAdmin ? "Admin" : (subCreate || subManage ? "Sub-admin" : "User"); await admin.from("audit").insert({ request_id: null, ref: null, actor_id: who.user.id, actor_name: caller.name, action: "Created an account", detail: `${name} <${email}> · ${roleLabel} · ${role}${dept ? " · " + dept : ""} · ${mode}`, }); return reply(200, { id: userId, email, mode, tempPassword }); });
