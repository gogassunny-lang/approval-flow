// SETU — "notify" edge function // Triggered by a database webhook on INSERT into public.notifications. // Sends the row as an email (Brevo) and a phone push (OneSignal via Median), // then records what was sent on the row. Either channel can be left // unconfigured; the other still works. // // Secrets (Edge Functions → Secrets): // APP_URL https://gogassunny-lang.github.io/approval-flow/ // MAIL_FROM_NAME SETU // Email, pick ONE of: // GMAIL_USER + GMAIL_APP_PASSWORD send straight from a Gmail account (app password, 2-Step Verification on) // BREVO_API_KEY + MAIL_FROM_EMAIL send through Brevo (sender verified there) // ONESIGNAL_APP_ID from onesignal.com → your app → Settings → Keys & IDs // ONESIGNAL_REST_API_KEY same page // SUPABASE_URL and SUPABASE_SERVICE_ROLE_KEY are provided automatically. import { createClient } from "npm:@supabase/supabase-js@2"; import { SMTPClient } from "https://deno.land/x/denomailer@1.6.0/mod.ts"; const env = (k: string) => Deno.env.get(k)?.trim() || ""; const esc = (s: string) => String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" }[c]!)); Deno.serve(async (req) => { let payload: any; try { payload = await req.json(); } catch { return new Response("bad json", { status: 400 }); } if (payload?.type !== "INSERT" || payload?.table !== "notifications" || !payload?.record?.id) { return new Response("ignored", { status: 200 }); } const n = payload.record; const sb = createClient(env("SUPABASE_URL"), env("SUPABASE_SERVICE_ROLE_KEY")); const { data: person } = await sb.from("profiles").select("name,email,active").eq("id", n.user_id).single(); if (!person || !person.active) return new Response("no active recipient", { status: 200 }); const appUrl = env("APP_URL") || "https://gogassunny-lang.github.io/approval-flow/"; const link = n.request_id ? `${appUrl}#detail/${n.request_id}` : appUrl; const firstName = String(person.name || "").split(" ")[0] || "there"; const errors: string[] = []; let email_sent_at: string | null = null; let push_sent_at: string | null = null; // ---------- email ---------- const gmailUser = env("GMAIL_USER"), gmailPass = env("GMAIL_APP_PASSWORD"); const brevoKey = env("BREVO_API_KEY"), brevoFrom = env("MAIL_FROM_EMAIL"); const fromName = env("MAIL_FROM_NAME") || "SETU"; if ((gmailUser && gmailPass) || (brevoKey && brevoFrom)) { if (person.email) { const html = `
SETU

Hello ${esc(firstName)},

${esc(n.title)}

${n.body ? `

${esc(n.body)}

` : ""} Open in SETU

You are receiving this because the request is on your desk in SETU. Decisions are signed in the app with your daily PIN; nothing can be approved from this email.

`; const text = `${n.title}\n\n${n.body || ""}\n\nOpen in SETU: ${link}`; if (gmailUser && gmailPass) { // straight from the Gmail account, over TLS on 465 try { const client = new SMTPClient({ connection: { hostname: "smtp.gmail.com", port: 465, tls: true, auth: { username: gmailUser, password: gmailPass } } }); await client.send({ from: `${fromName} <${gmailUser}>`, to: `${person.name} <${person.email}>`, subject: n.title, content: text, html }); await client.close(); email_sent_at = new Date().toISOString(); } catch (e) { errors.push(`gmail: ${String(e).slice(0, 200)}`); } } else { try { const r = await fetch("https://api.brevo.com/v3/smtp/email", { method: "POST", headers: { "api-key": brevoKey, "content-type": "application/json", accept: "application/json" }, body: JSON.stringify({ sender: { name: fromName, email: brevoFrom }, to: [{ email: person.email, name: person.name }], subject: n.title, htmlContent: html, textContent: text, tags: ["setu", String(n.kind)] }), }); if (r.ok) email_sent_at = new Date().toISOString(); else errors.push(`brevo ${r.status}: ${(await r.text()).slice(0, 200)}`); } catch (e) { errors.push(`brevo: ${String(e).slice(0, 200)}`); } } } } // ---------- push via OneSignal (the Median app logs each person in by their SETU user id) ---------- const osApp = env("ONESIGNAL_APP_ID"), osKey = env("ONESIGNAL_REST_API_KEY"); if (osApp && osKey) { try { const r = await fetch("https://api.onesignal.com/notifications?c=push", { method: "POST", headers: { Authorization: `Key ${osKey}`, "content-type": "application/json", accept: "application/json" }, body: JSON.stringify({ app_id: osApp, target_channel: "push", // match both the clean user id AND the older wrapped form some devices // registered with ({"externalId":""}), so every phone is reached include_aliases: { external_id: [n.user_id, JSON.stringify({ externalId: n.user_id })] }, headings: { en: n.title }, contents: { en: (n.body || "Open SETU to see the request.").split("\n")[0].slice(0, 180) }, url: link, android_channel_id: env("ONESIGNAL_ANDROID_CHANNEL") || undefined, data: { request_id: n.request_id, kind: n.kind }, }), }); const body = await r.json().catch(() => ({})); // "no recipients" just means this person has not installed the app; not an error if (r.ok && !(body?.errors && String(body.errors).includes("subscribed"))) push_sent_at = new Date().toISOString(); else if (!r.ok) errors.push(`onesignal ${r.status}: ${JSON.stringify(body).slice(0, 200)}`); } catch (e) { errors.push(`onesignal: ${String(e).slice(0, 200)}`); } } await sb.from("notifications").update({ email_sent_at, push_sent_at, error: errors.join(" | ") || null }).eq("id", n.id); return new Response(JSON.stringify({ email: !!email_sent_at, push: !!push_sent_at, errors }), { headers: { "content-type": "application/json" }, }); });
